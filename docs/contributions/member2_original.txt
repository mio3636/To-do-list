# 2. JSON 存储具体实现 (对应图片：管理文件路径、读取和校验 JSON)
# ==========================================
class JsonTaskRepository(TaskRepository):
    def __init__(self, filepath="tasks.json"):
        self.filepath = filepath

    def load(self) -> list:
        """读取并校验 JSON 数据"""
        if not os.path.exists(self.filepath):
            return [] # 文件不存在则返回空列表
            
        try:
            with open(self.filepath, 'r', encoding='utf-8') as f:
                data = json.load(f)
                # 简单的数据校验：确保读取出来的是列表
                if not isinstance(data, list):
                    print("Warning: Data format error, expected a list. Returning empty.")
                    return []
                return data
        except json.JSONDecodeError:
            print("Warning: JSON file is corrupted. Returning empty list.")
            return []
        except Exception as e:
            print(f"Error loading tasks: {e}")
            return []

    def save(self, tasks: list) -> None:
        """安全保存数据到 JSON"""
        try:
            # 使用临时文件+重命名的方式，防止保存过程中程序崩溃导致文件损坏
            temp_path = self.filepath + ".tmp"
            with open(temp_path, 'w', encoding='utf-8') as f:
                json.dump(tasks, f, indent=4, ensure_ascii=False)
            # 原子替换
            os.replace(temp_path, self.filepath)
        except Exception as e:
            print(f"Error saving tasks: {e}")

def add_task(tasks, desc, prio, due, cat):
    task = {
        "id": len(tasks) + 1 if not tasks else max(t["id"] for t in tasks) + 1,
        "description": desc,
        "priority": prio,
        "due_date": due,
        "category": cat,
        "status": "in progress"
    }
    tasks.append(task)
    print(f"Task '{desc}' added.")

def view_tasks(tasks):
    if not tasks:
        print("No tasks found.")
        return
    for i, t in enumerate(tasks, 1):
        print(f"[{i}] {t.get('description')} | Status: {t.get('status')} | Due: {t.get('due_date')}")

def mark_task_status(tasks, idx, status):
    if 0 < idx <= len(tasks):
        tasks[idx - 1]["status"] = status
        print(f"Task {idx} status updated to {status}.")
    else:
        print("Invalid index.")

def delete_task(tasks, idx):
    if 0 < idx <= len(tasks):
        deleted = tasks.pop(idx - 1)
        print(f"Deleted task: {deleted['description']}")
    else:
        print("Invalid index.")
